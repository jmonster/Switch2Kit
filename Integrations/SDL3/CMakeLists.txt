cmake_minimum_required(VERSION 3.24)
if(TARGET Switch2Kit::SDL3)
  return()
endif()
if(NOT TARGET SDL3::SDL3)
  find_package(SDL3 3.2 CONFIG REQUIRED)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../CMake" "${CMAKE_CURRENT_BINARY_DIR}/c-api")
add_library(Switch2KitSDL3 STATIC Switch2KitSDL3.cpp)
add_library(Switch2Kit::SDL3 ALIAS Switch2KitSDL3)
target_compile_features(Switch2KitSDL3 PUBLIC cxx_std_17)
target_include_directories(Switch2KitSDL3 PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(Switch2KitSDL3 PUBLIC Switch2Kit::C SDL3::SDL3)
set_target_properties(Switch2KitSDL3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
