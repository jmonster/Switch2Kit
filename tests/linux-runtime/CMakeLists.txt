cmake_minimum_required(VERSION 3.24)
project(Switch2KitLinuxRuntime LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This loader regression requires Linux")
endif()
get_filename_component(_default_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SWITCH2KIT_SOURCE_DIR "${_default_root}" CACHE PATH "Switch2Kit source checkout")
add_subdirectory("${SWITCH2KIT_SOURCE_DIR}/Integrations/CMake" switch2kit)
include(GNUInstallDirs)
foreach(_target IN ITEMS imported_consumer absolute_consumer)
  add_executable(${_target} main.cpp)
  target_compile_features(${_target} PRIVATE cxx_std_17)
  target_compile_options(${_target} PRIVATE -Wall -Wextra -Werror -pedantic)
  target_include_directories(${_target} PRIVATE "${SWITCH2KIT_SOURCE_DIR}/Sources/Switch2KitCABI/include")
  target_link_libraries(${_target} PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(${_target} Switch2KitCBuild)
  switch2kit_install_linux(${_target})
  install(TARGETS ${_target} RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endforeach()
target_link_libraries(imported_consumer PRIVATE Switch2Kit::C)
# Exercise hosts that link the generated library by absolute path. RPATH cannot
# override a slash-containing DT_NEEDED; the library must supply a real SONAME.
target_link_libraries(absolute_consumer PRIVATE "$<TARGET_FILE:Switch2Kit::C>")
